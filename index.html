function haystack(e) {

  const s = SUBJ[e.subject]; const g = (s.groups || []).find(x => x.id === e.group);

  return norm([e.title, s.name, s.fullName, (s.aliases || []).join(' '), e.exam, yearText(e), e.yearFrom != null ? `${e.yearFrom}-${e.yearTo}` : '', e.level, g && g.label, catLabel(s.category)].join(' '));

}

function matchQ(e, q) { const t = norm(q).split(/\s+/).filter(Boolean); if (!t.length) return true; const h = haystack(e); return t.every(x => h.includes(x)); }

function filterExams(list, f = {}) {

  return list.filter(e =>

    (!f.subject || f.subject === 'all' || e.subject === f.subject) &&

    (!f.section || f.section === 'all' || e.section === f.section) &&

    (!f.group || f.group === 'all' || e.group === f.group) &&

    (!f.exam || f.exam === 'all' || e.exam === f.exam) &&

    (!f.year || f.year === 'all' || covers(e, +f.year)) &&

    (f.fileType !== 'answer' || !!(e.files && e.files.answer)) &&

    matchQ(e, f.q));

}

const ORDER_IDX = new Map(D.exams.map((e, i) => [e.id, i]));

function sortExams(list, sort) {

  return [...list].sort((a, b) => {

    if (isAvail(a) !== isAvail(b)) return isAvail(a) ? -1 : 1;

    if (sort === 'popular') { const pa = a.popular ?? 999, pb = b.popular ?? 999; if (pa !== pb) return pa - pb; }

    return (yMax(b) - yMax(a)) || (ORDER_IDX.get(a.id) - ORDER_IDX.get(b.id));

  });

}

function statValue(v) {

  if (v === 'auto:subjects') return String(D.subjects.length);

  if (v === 'auto:items') return String(D.exams.filter(isAvail).length);

  return v;

}

/* Hooks */

function parseHash() {

  const raw = location.hash.replace(/^#/, '') || '/';

  const [path, qs] = raw.split('?');

  return { key: raw, path: path || '/', params: Object.fromEntries(new URLSearchParams(qs || '')) };

}

function useHashRoute() {

  const [r, setR] = useState(parseHash);

  useEffect(() => { const f = () => setR(parseHash()); window.addEventListener('hashchange', f); return () => window.removeEventListener('hashchange', f); }, []);

  return r;

}

function navigate(path, params) {

  const qs = params && Object.keys(params).length ? '?' + new URLSearchParams(params).toString() : '';

  location.hash = path + qs;

}

function sameHashClick(e, href) {

  if (href && href.startsWith('#') && location.hash === href) {

    e.preventDefault();

    const to = new URLSearchParams(href.split('?')[1] || '').get('to');

    if (to) document.getElementById(to)?.scrollIntoView({ behavior: 'smooth', block: 'start' }); else window.scrollTo({ top: 0, behavior: 'smooth' });

  }

}

function useScrolled(y = 8) {

  const [s, setS] = useState(false);

  useEffect(() => { const f = () => setS(window.scrollY > y); f(); window.addEventListener('scroll', f, { passive: true }); return () => window.removeEventListener('scroll', f); }, [y]);

  return s;

}

function useScrolling() {

  const [s, setS] = useState(false);

  useEffect(() => { let t; const f = () => { setS(true); clearTimeout(t); t = setTimeout(() => setS(false), 900); }; window.addEventListener('scroll', f, { passive: true }); return () => { window.removeEventListener('scroll', f); clearTimeout(t); }; }, []);

  return s;

}

function useDarkMode() {

  const [dark, setDark] = useState(() => document.documentElement.classList.contains('dark'));

  useEffect(() => { document.documentElement.classList.toggle('dark', dark); try { localStorage.setItem('smp_theme', dark ? 'dark' : 'light'); } catch (e) {} }, [dark]);

  return [dark, setDark];

}

function Reveal({ as: Tag = 'div', className, delay = 0, children, ...rest }) {

  const ref = useRef(null); const [inView, setIn] = useState(false);

  useEffect(() => {

    const el = ref.current; if (!el) return;

    if (!('IntersectionObserver' in window)) { setIn(true); return; }

    const io = new IntersectionObserver(([en]) => { if (en.isIntersecting) { setIn(true); io.disconnect(); } }, { rootMargin: '0px 0px -6% 0px' });

    io.observe(el); return () => io.disconnect();

  }, []);

  return <Tag ref={ref} className={cx('reveal', inView && 'is-in', className)} style={{ transitionDelay: `${delay}ms` }} {...rest}>{children}</Tag>;

}

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

function sanitizePhone(raw) {

  let d = (raw || '').replace(/\D/g, '');

  if (d.startsWith('66') && d.length >= 11) d = '0' + d.slice(2);

  return d.slice(0, 10);

}

function validateField(k, v) {

  const M = C.messages;

  switch (k) {

    case 'email': return !v.email.trim() ? M.emailRequired : !EMAIL_RE.test(v.email.trim()) ? M.email : '';

    case 'phone': return !v.phone ? M.phoneRequired : !/^0\d{9}$/.test(v.phone) ? M.phone : '';

    case 'everCourse': case 'prep': return v[k] ? '' : M.chooseOne;

    case 'exams': return v.exams.length ? '' : M.exams;

    case 'consent': return !v.consent ? M.consent : (v.consent === 'no' && !C.form.allowSubmitWithoutConsent) ? M.consentRequired : '';

    default: return '';

  }

}

function Combobox({ id, value, onChange, options, placeholder, allowCustom, customHint }) {

  const [open, setOpen] = useState(false); const [q, setQ] = useState(value || ''); const [active, setActive] = useState(0);

  useEffect(() => { setQ(value || ''); }, [value]);

  const t = norm(q.trim());

  const filtered = (t && !options.includes(q) ? options.filter(o => norm(o).includes(t)) : options).slice(0, 60);

  const list = [...filtered.map(o => ({ type: 'opt', v: o })), ...(allowCustom ? [{ type: 'custom' }] : [])];

  const choose = en => {

    if (!en) return;

    if (en.type === 'custom') { onChange(q.trim()); setOpen(false); document.getElementById(id)?.focus(); }

    else { onChange(en.v); setQ(en.v); setOpen(false); }

  };

  const onKey = e => {

    if (e.key === 'ArrowDown') { e.preventDefault(); setOpen(true); setActive(a => Math.min(a + 1, list.length - 1)); }

    else if (e.key === 'ArrowUp') { e.preventDefault(); setActive(a => Math.max(a - 1, 0)); }

    else if (e.key === 'Enter' && open) { e.preventDefault(); choose(list[active]); }

    else if (e.key === 'Escape' && open) { e.stopPropagation(); setOpen(false); }

  };

  const onBlur = () => setTimeout(() => {

    setOpen(false);

    if (allowCustom) onChange(q.trim());

    else if (!options.includes(q)) setQ(value || '');

  }, 120);

  return (

    <div className="relative">

      <input id={id} type="text" role="combobox" aria-expanded={open} aria-controls={`${id}-list`} aria-autocomplete="list"

        aria-activedescendant={open && list[active] ? `${id}-opt-${active}` : undefined}

        value={q} placeholder={placeholder} autoComplete="off" className="field select"

        onChange={e => { setQ(e.target.value); setOpen(true); setActive(0); }} onFocus={() => setOpen(true)} onClick={() => setOpen(true)} onKeyDown={onKey} onBlur={onBlur} />

      {open && (

        <ul id={`${id}-list`} role="listbox" className="absolute z-20 mt-1 max-h-60 w-full overflow-auto rounded-xl border border-line bg-card p-1 shadow-xl">

          {list.length === 0 && <li className="px-3 py-2.5 text-sm text-muted">ไม่พบตัวเลือก</li>}

          {list.map((en, i) => (

            <li key={i} id={`${id}-opt-${i}`} role="option" aria-selected={en.type === 'opt' && en.v === value}

              onMouseDown={e => { e.preventDefault(); choose(en); }} onMouseEnter={() => setActive(i)}

              className={cx('cursor-pointer rounded-lg px-3 py-2.5 text-[15px]', i === active && 'bg-pblue', en.type === 'custom' ? 'mt-1 border-t border-line font-semibold text-link' : 'text-ink')}>

              {en.type === 'custom' ? (q.trim() ? `ไม่พบโรงเรียนของฉัน → ใช้ "${q.trim()}"` : customHint) : en.v}

            </li>

          ))}

        </ul>

      )}

    </div>

  );

}

function App() {

  const route = useHashRoute();

  const [dark, setDark] = useDarkMode();

  const init = useMemo(() => store.load(), []);

  const [registered, setRegistered] = useState(init.reg || {});

  const [profile, setProfile] = useState(init.profile || null);

  const [modal, setModal] = useState(null);

  const [toasts, setToasts] = useState([]);


  const toast = useCallback((text, tone = 'success') => {

    const id = Math.random().toString(36).slice(2);

    setToasts(t => [...t, { id, text, tone }]);

    setTimeout(() => setToasts(t => t.filter(x => x.id !== id)), 4500);

  }, []);

  useEffect(() => { store.save({ reg: registered, profile }); }, [registered, profile]);


  const deliver = useCallback(async (item, kind) => {

    try {

      const url = await resolveFileUrl(item, kind);

      if (!url || url === '#') { toast(`เดโม: ยังไม่ได้ใส่ลิงก์ไฟล์จริงของ "${item.title}"`, 'info'); return; }

      const a = document.createElement('a'); a.href = url; a.target = '_blank'; a.rel = 'noopener'; a.download = '';

      document.body.appendChild(a); a.click(); a.remove();

    } catch (e) { toast('ดาวน์โหลดไม่สำเร็จ กรุณาลองอีกครั้ง', 'error'); }

  }, [toast]);


  const isUnlocked = useCallback(id => !!registered[id], [registered]);

  const requestDownload = useCallback((item, kind) => {

    if (!isAvail(item)) return;

    if (registered[item.subject]) deliver(item, kind);

    else setModal({ subjectId: item.subject, pending: { item, kind } });

  }, [registered, deliver]);

  const openRegister = useCallback(id => setModal({ subjectId: id, pending: null }), []);

  const onRegistered = useCallback((subjectId, values) => {

    setRegistered(r => ({ ...r, [subjectId]: true }));   // ปลดล็อกทั้งวิชา

    setProfile(pickProfile(values));                       // prefill วิชาถัดไป

    const p = modal && modal.pending;

    setModal(null);

    toast(C.messages.success);

    if (p) setTimeout(() => deliver(p.item, p.kind), 150);

  }, [modal, deliver, toast]);


  useEffect(() => {

    if (route.params.to) setTimeout(() => document.getElementById(route.params.to)?.scrollIntoView({ behavior: 'smooth', block: 'start' }), 80);

    else window.scrollTo(0, 0);

  }, [route.key]);


  const m = route.path.match(/^\/subject\/([\w-]+)/);

  let page;

  if (m) page = SUBJ[m[1]] ? <ExamsPage key={route.key} subjectId={m[1]} params={route.params} /> : <NotFound />;

  else if (route.path === '/exams') page = <ExamsPage key={route.key} subjectId={null} params={route.params} />;

  else if (route.path === '/') page = <Home />;

  else page = <NotFound />;


  return (

    <AppCtx.Provider value={{ isUnlocked, requestDownload, openRegister }}>

      <a href="#main" className="sr-only focus:not-sr-only focus:fixed focus:left-3 focus:top-3 focus:z-[200] focus:rounded-lg focus:bg-card focus:px-4 focus:py-2" onClick={e => { e.preventDefault(); document.getElementById('main')?.focus(); }}>ข้ามไปเนื้อหาหลัก</a>

      <Navbar dark={dark} setDark={setDark} route={route} />

      <main id="main" tabIndex={-1} className="outline-none">{page}</main>

      <Footer />

      <FloatingButtons hidden={!!modal} />

      {modal && <RegisterModal key={modal.subjectId} subjectId={modal.subjectId} profile={profile} onClose={() => setModal(null)} onSuccess={onRegistered} />}

      <Toasts items={toasts} />

    </AppCtx.Provider>

  );

}


ReactDOM.createRoot(document.getElementById('root')).render(<App />);
